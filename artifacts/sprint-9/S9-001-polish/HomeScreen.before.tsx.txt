import { useState } from 'react';
import { Image, Modal, Pressable, ScrollView, StyleSheet, Text, useWindowDimensions, View } from 'react-native';
import { SafeAreaView } from 'react-native-safe-area-context';
import type { AppSnapshot } from '../application/app-runtime.ts';
import { homeScreenModel } from '../application/ui-model.ts';
import FinniHomeScene, { type HomeReaction } from './FinniHomeScene.tsx';
import { OBJECT_SOURCES, goalSource } from './room-assets.ts';
import type { SceneRect } from './home-scene-layout.ts';

export type HomeScreenProps = Readonly<{
  snapshot: AppSnapshot; busy: boolean; notice: string | null; scenePaused: boolean;
  motionEnabled: boolean; reaction: HomeReaction | null; lessonTitle: string | null; demo: boolean;
  onReactionFinished: (id: number) => void; onReactionCancelled: (id: number) => void;
  onDismissNotice: () => void; onOpenDay: () => void; onResults: () => void; onEditPet: () => void;
  onHelp: () => void; onLesson: () => void; onSection: (title: string) => void;
}>;

const HOME_ICONS = {
  home: require('../../assets/ui/home/home.png'),
  progress: require('../../assets/ui/home/progress.png'),
  adult: require('../../assets/ui/home/adult.png'),
  help: require('../../assets/ui/home/help.png'),
};
function RouteButton({ label, onPress, selected = false, icon, compact = false }: Readonly<{ label: string; onPress: () => void; selected?: boolean; icon?: keyof typeof HOME_ICONS; compact?: boolean }>) {
  return <Pressable accessibilityRole="button" accessibilityState={{ selected }} onPress={onPress} style={({ pressed }) => [styles.route, icon && styles.railRoute, compact && styles.compactRoute, pressed && styles.pressed]}>{icon && <Image accessible={false} source={HOME_ICONS[icon]} style={[styles.routeIcon, compact && styles.compactRouteIcon]} />}<Text style={styles.routeText}>{label}</Text></Pressable>;
}

/** Home owns presentation only. All finance routes retain their confirmation screens. */
export default function HomeScreen(props: HomeScreenProps) {
  const viewport = useWindowDimensions();
  const large = viewport.fontScale > 1.2;
  const profile = props.snapshot.profile!;
  const life = props.snapshot.lifecycle!;
  const model = homeScreenModel(profile, life);
  const goal = props.snapshot.commerce?.selectedGoal;
  const purchases = props.snapshot.commerce?.purchases ?? [];
  const [menu, setMenu] = useState(false);
  const [skipReactionId, setSkipReactionId] = useState<number | null>(null);
  const canSkip = props.reaction?.skippable && skipReactionId !== props.reaction.id;
  const [size, setSize] = useState({ width: viewport.width, height: viewport.height });
  const [region, setRegion] = useState<SceneRect>({ x: 6, y: 220, width: 100, height: 100 });
  const [copyHeight, setCopyHeight] = useState(112);
  const [sceneRect, setSceneRect] = useState<SceneRect>({ x: 12, y: 150, width: 336, height: 300 });
  const [failedObjects, setFailedObjects] = useState<readonly string[]>([]);
  const stateWord = (slot: string) => purchases.some(({ item }) => item.slot === slot) ? 'готово'
    : life.state === 'CLOSED' || life.state === 'WAITING' ? 'нет' : 'выберем';
  const food = `Еда: ${stateWord('food')}`;
  const care = `Уход: ${stateWord('care')}`;
  const mood = props.reaction?.expression === 'happy' ? 'Радостно' : props.reaction?.expression === 'thoughtful' ? 'Задумчиво'
    : props.reaction?.expression === 'inspired' ? 'Вдохновлённо' : 'Спокойно';
  const allGoals = (props.snapshot.commerce?.claimedGoalIds.length ?? 0) === 3;
  const goalName = goal?.name ?? (allGoals ? 'Мечты этого набора получены' : 'Выбери мечту');
  const remaining = goal ? Math.max(0, goal.cost - life.savings) : null;
  const primary = () => {
    if (model.action.route === 'open-day') props.onOpenDay();
    else if (model.action.route === 'plan') props.onSection('План');
    else if (model.action.route === 'day' || model.action.route === 'results') props.onResults();
  };
  const compactAction = { READY: 'Начать день', DRAFT: 'К плану', ACTIVE: 'Мой день', CLOSED: 'Итоги дня', WAITING: 'День позже', STORAGE_ERROR: 'Повторить', NO_PROFILE: 'Питомец' }[life.state];
  const navigate = (action: () => void) => { setMenu(false); action(); };
  const text = [styles.text, large && styles.largeText];
  const small = [styles.small, large && styles.largeSmall];
  const button = <Pressable accessibilityRole="button" accessibilityLabel={props.busy ? 'Подождите' : model.action.label} accessibilityState={{ disabled: props.busy || !model.action.enabled }} disabled={props.busy || !model.action.enabled} onPress={primary} style={[styles.primary, large && styles.largePrimary, (!model.action.enabled || props.busy) && styles.disabled]} testID="home-primary"><Text style={[styles.primaryText, large && styles.largePrimaryText]}>{props.busy ? 'Подождите…' : large ? compactAction : model.action.label}</Text></Pressable>;
  const lesson = <Pressable accessibilityRole="button" accessibilityHint="Открывает список занятий" onPress={props.onLesson} style={[styles.lesson, large && styles.largeLesson]} testID="home-lesson"><Text style={small}>Занятие</Text><Text style={[text, styles.bold]}>{props.lessonTitle ?? 'Выбери занятие'}</Text></Pressable>;
  const state = <View style={[styles.state, large && styles.largeState]} testID="home-care"><Text style={large ? small : styles.statusText}>{food}</Text><Text style={large ? small : styles.statusText}>{care}</Text><Text accessibilityLabel={`Настроение: ${mood}`} style={large ? small : styles.statusText}>{mood}</Text></View>;

  return <SafeAreaView style={styles.safe}>
    <View style={styles.home} onLayout={({ nativeEvent: { layout } }) => setSize(layout)} testID={large ? 'home-large-summary' : 'home-v42'}>
      <View pointerEvents="none" style={StyleSheet.absoluteFill} importantForAccessibility="no-hide-descendants">
        <FinniHomeScene key={`${profile.shapeId}-${profile.patternId}`} accessibilityLabel={`${profile.name}. ${food}. ${care}. ${mood}`} careLabel={`${food}. ${care}`} height={size.height} paused={props.scenePaused || menu || Boolean(props.notice)} motionEnabled={props.motionEnabled} stage={life.petStage} appearance={profile} reaction={props.reaction} onReactionFinished={props.onReactionFinished} onReactionCancelled={props.onReactionCancelled} showCaption={false} roomOpacity={large ? .14 : 1} fullscreen petRegion={large ? region : { x: sceneRect.x + (size.height < 700 ? 70 : 4), y: sceneRect.y + copyHeight + 8, width: Math.max(80, sceneRect.width - (size.height < 700 ? 148 : 92)), height: Math.max(48, sceneRect.height - copyHeight - (size.height < 700 ? 12 : 76)) }} hideSkip skipReactionId={skipReactionId} />
      </View>
      <View testID="home-hud" style={[styles.hud, large && styles.largeHud, { height: size.height }]} accessibilityElementsHidden={menu} importantForAccessibility={menu ? 'no-hide-descendants' : 'auto'} pointerEvents={menu ? 'none' : 'box-none'}>
        <View style={[styles.summary, large && styles.largeSummary]} testID="home-finances">
          {large && !goal && <Text style={small}>Монеты</Text>}
          <View style={[styles.money, large && styles.largeMoney]}>{[['Доступно', life.available], ['Копилка', life.savings]].map(([label, value]) => <View key={label} accessible accessibilityLabel={label + ': ' + value + ' монет'} style={[styles.metric, large && styles.largeMetric]}>{!large && <Image accessible={false} source={OBJECT_SOURCES[label === 'Доступно' ? 'OBJ-COIN' : 'OBJ-CHEST']} style={styles.moneyIcon} />}<View style={large ? styles.largeMetricContent : styles.metricContent}><Text style={small}>{large ? label : label + ', монет'}</Text><Text style={[styles.amount, large && styles.largeAmount]}>{value}</Text></View></View>)}</View>
          <Pressable accessibilityRole="button" accessibilityHint="Открывает цель и накопления" onPress={() => props.onSection('Копилка')} style={[styles.goal, large && styles.largeGoal]} testID="home-goal">
            <View style={styles.goalHeading}>{!large && <Image accessible={false} source={goalSource(goal?.id ?? null) ?? OBJECT_SOURCES['OBJ-GOAL-DISPLAY']} resizeMode="contain" style={styles.goalIcon} />}<Text style={[text, styles.bold, styles.goalTitle]}>{'Цель: ' + goalName}</Text></View>
            {goal && (large ? <><Text style={small}>{life.savings} из {goal.cost} монет</Text><Text style={small}>Осталось {remaining}</Text></> : <><Text style={small}>{life.savings} из {goal.cost} монет · Осталось {remaining}</Text><View accessibilityRole="progressbar" accessibilityValue={{ min: 0, max: goal.cost, now: Math.min(life.savings, goal.cost) }} style={styles.track}><View style={[styles.fill, { width: `${Math.min(100, life.savings / goal.cost * 100)}%` }]} /></View></>)}
          </Pressable>
        </View>
        {large ? <>
          <View style={styles.largePetRow} onLayout={({ nativeEvent: { layout } }) => setRegion({ x: layout.x, y: layout.y, width: 104, height: layout.height })}>
            <Pressable accessibilityRole="button" accessibilityLabel={`${profile.name}, стадия ${life.petStage}. Изменить питомца`} onPress={props.onEditPet} style={styles.petTarget} testID="home-pet-target" />{state}
          </View>{lesson}
          <View testID="home-footer" style={styles.largeFooter}><Pressable accessibilityRole="button" accessibilityLabel={canSkip ? "Пропустить анимацию" : "Разделы"} onPress={() => canSkip ? setSkipReactionId(props.reaction!.id) : setMenu(true)} style={styles.sections} testID="home-sections"><Text style={small}>{canSkip ? "Пропуск" : "Разделы"}</Text></Pressable>{button}</View>
        </> : <>
          <View style={styles.sceneSpace} onLayout={({ nativeEvent: { layout } }) => setSceneRect(layout)}>
            <View style={styles.sceneCopy} onLayout={({ nativeEvent: { layout } }) => setCopyHeight(layout.height)}><View style={styles.inlineState}>{state}</View>{lesson}</View>
            <Pressable accessibilityRole="button" accessibilityLabel={`${profile.name}, стадия ${life.petStage}. ${mood}. Изменить питомца`} onPress={props.onEditPet} style={[styles.ordinaryPetTarget, { top: copyHeight + 8 }, size.height < 700 && { left: 70, right: 78, bottom: 0 }]} />
            <View style={styles.rightRail}><RouteButton compact={size.height < 700} icon="progress" label="Прогресс" onPress={() => props.onSection('Прогресс')} /><RouteButton compact={size.height < 700} icon="adult" label={'Для\nвзрослого'} onPress={() => props.onSection('Для взрослого')} /><RouteButton compact={size.height < 700} icon="help" label={canSkip ? 'Пропустить' : 'Как играть'} onPress={canSkip ? () => setSkipReactionId(props.reaction!.id) : props.onHelp} /></View>
              <View pointerEvents="box-none" style={styles.objects}>{[
                ['OBJ-PLANNER', 'Планер', 'План'], ['OBJ-CARE', 'Забота', 'Покупки'], ['OBJ-CHEST', 'Копилка', 'Копилка'],
              ].map(([id, label, route]) => <Pressable key={id} accessibilityRole="button" accessibilityLabel={label} onPress={() => props.onSection(route!)} style={[styles.object, id === 'OBJ-PLANNER' ? styles.plannerAnchor : id === 'OBJ-CARE' ? styles.careAnchor : id === 'OBJ-GOAL-DISPLAY' ? styles.goalAnchor : styles.chestAnchor, id === 'OBJ-CARE' && size.height < 700 && { left: 0, right: undefined }]} testID={`room-object-${id}`}>
                {id === 'OBJ-PLANNER' && <View pointerEvents="none" style={styles.table}><View style={styles.tableLeftLeg} /><View style={styles.tableRightLeg} /></View>}
                {!failedObjects.includes(id!) && <Image accessible={false} source={id === 'OBJ-GOAL-DISPLAY' ? goalSource(goal?.id ?? null) ?? OBJECT_SOURCES[id!] : OBJECT_SOURCES[id!]} resizeMode="contain" onError={() => setFailedObjects((old) => [...old, id!])} style={[styles.objectImage, id === 'OBJ-PLANNER' && styles.plannerImage]} />}<Text style={[styles.objectLabel, id === 'OBJ-PLANNER' && styles.plannerLabel]}>{label}</Text>
              </Pressable>)}</View>
          </View>
          <View testID="home-footer" style={styles.footer}>{button}<View style={styles.nav} accessibilityRole="tablist">{['Домик', 'План', 'Покупки', 'Копилка'].map((label) => <Pressable key={label} accessibilityRole="tab" accessibilityState={{ selected: label === 'Домик' }} onPress={() => label !== 'Домик' && props.onSection(label)} style={[styles.navItem, label === 'Домик' && styles.navSelected]}>{label === 'Домик' ? <Image accessible={false} source={HOME_ICONS.home} style={styles.navIcon} /> : <Image accessible={false} source={OBJECT_SOURCES[label === 'План' ? 'OBJ-PLANNER' : label === 'Покупки' ? 'OBJ-CARE' : 'OBJ-CHEST']} style={styles.navIcon} />}<Text style={[styles.small, label === 'Домик' && styles.bold]}>{label}</Text></Pressable>)}</View></View>
        </>}
      </View>
      <Modal visible={menu} onRequestClose={() => setMenu(false)} animationType="none" accessibilityViewIsModal>
        <SafeAreaView style={styles.menu}><ScrollView contentContainerStyle={styles.menuContent}><Text style={[styles.text, styles.bold]}>{profile.name}</Text><Text style={styles.text}>{model.dayLabel}{props.demo ? ' · Демо' : ''}</Text><RouteButton label="Вернуться в домик" onPress={() => setMenu(false)} />{['План', 'Покупки', 'Копилка', 'Прогресс', 'Для взрослого'].map((label) => <RouteButton key={label} label={label} onPress={() => navigate(() => props.onSection(label))} />)}<RouteButton label="Занятия" onPress={() => navigate(props.onLesson)} /><RouteButton label="О питомце" onPress={() => navigate(props.onEditPet)} /><RouteButton label="Как играть" onPress={() => navigate(props.onHelp)} /></ScrollView></SafeAreaView>
      </Modal>
      <Modal visible={Boolean(props.notice)} transparent onRequestClose={props.onDismissNotice}><View style={styles.noticeOverlay}><View style={styles.summary}><Text accessibilityLiveRegion="polite" style={styles.text}>{props.notice}</Text><RouteButton label="Понятно" onPress={props.onDismissNotice} /></View></View></Modal>
    </View>
  </SafeAreaView>;
}
