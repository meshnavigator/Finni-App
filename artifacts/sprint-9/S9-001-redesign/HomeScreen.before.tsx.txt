import { useState } from 'react';
import { Image, Modal, Pressable, ScrollView, StyleSheet, Text, useWindowDimensions, View } from 'react-native';
import { SafeAreaView } from 'react-native-safe-area-context';
import type { AppSnapshot } from '../application/app-runtime.ts';
import { homeScreenModel } from '../application/ui-model.ts';
import FinniHomeScene, { type HomeReaction } from './FinniHomeScene.tsx';
import { OBJECT_SOURCES, goalSource } from './room-assets.ts';
import type { SceneRect } from './home-scene-layout.ts';

export type HomeScreenProps = Readonly<{
  snapshot: AppSnapshot; busy: boolean; notice: string | null; scenePaused: boolean;
  motionEnabled: boolean; reaction: HomeReaction | null; lessonTitle: string | null; demo: boolean;
  onReactionFinished: (id: number) => void; onReactionCancelled: (id: number) => void;
  onDismissNotice: () => void; onOpenDay: () => void; onResults: () => void; onEditPet: () => void;
  onHelp: () => void; onLesson: () => void; onSection: (title: string) => void;
}>;

function RouteButton({ label, onPress, selected = false }: Readonly<{ label: string; onPress: () => void; selected?: boolean }>) {
  return <Pressable accessibilityRole="button" accessibilityState={{ selected }} onPress={onPress} style={({ pressed }) => [styles.route, pressed && styles.pressed]}><Text style={styles.routeText}>{label}</Text></Pressable>;
}

/** Home owns presentation only. All finance routes retain their confirmation screens. */
export default function HomeScreen(props: HomeScreenProps) {
  const viewport = useWindowDimensions();
  const large = viewport.fontScale > 1.2;
  const profile = props.snapshot.profile!;
  const life = props.snapshot.lifecycle!;
  const model = homeScreenModel(profile, life);
  const goal = props.snapshot.commerce?.selectedGoal;
  const purchases = props.snapshot.commerce?.purchases ?? [];
  const [menu, setMenu] = useState(false);
  const [skipReactionId, setSkipReactionId] = useState<number | null>(null);
  const canSkip = props.reaction?.skippable && skipReactionId !== props.reaction.id;
  const [size, setSize] = useState({ width: viewport.width, height: viewport.height });
  const [region, setRegion] = useState<SceneRect>({ x: 6, y: 220, width: 100, height: 100 });
  const [failedObjects, setFailedObjects] = useState<readonly string[]>([]);
  const stateWord = (slot: string) => purchases.some(({ item }) => item.slot === slot) ? 'готово'
    : life.state === 'CLOSED' || life.state === 'WAITING' ? 'нет' : 'выберем';
  const food = `Еда: ${stateWord('food')}`;
  const care = `Уход: ${stateWord('care')}`;
  const mood = props.reaction?.expression === 'happy' ? 'Радостно' : props.reaction?.expression === 'thoughtful' ? 'Задумчиво'
    : props.reaction?.expression === 'inspired' ? 'Вдохновлённо' : 'Спокойно';
  const allGoals = (props.snapshot.commerce?.claimedGoalIds.length ?? 0) === 3;
  const goalName = goal?.name ?? (allGoals ? 'Мечты этого набора получены' : 'Выбери мечту');
  const remaining = goal ? Math.max(0, goal.cost - life.savings) : null;
  const primary = () => {
    if (model.action.route === 'open-day') props.onOpenDay();
    else if (model.action.route === 'plan') props.onSection('План');
    else if (model.action.route === 'day' || model.action.route === 'results') props.onResults();
  };
  const compactAction = { READY: 'Начать день', DRAFT: 'К плану', ACTIVE: 'Мой день', CLOSED: 'Итоги дня', WAITING: 'День позже', STORAGE_ERROR: 'Повторить', NO_PROFILE: 'Питомец' }[life.state];
  const navigate = (action: () => void) => { setMenu(false); action(); };
  const text = [styles.text, large && styles.largeText];
  const small = [styles.small, large && styles.largeSmall];
  const button = <Pressable accessibilityRole="button" accessibilityLabel={props.busy ? 'Подождите' : model.action.label} accessibilityState={{ disabled: props.busy || !model.action.enabled }} disabled={props.busy || !model.action.enabled} onPress={primary} style={[styles.primary, large && styles.largePrimary, (!model.action.enabled || props.busy) && styles.disabled]} testID="home-primary"><Text style={[styles.primaryText, large && styles.largeText]}>{props.busy ? 'Подождите…' : large ? compactAction : model.action.label}</Text></Pressable>;
  const lesson = <Pressable accessibilityRole="button" accessibilityHint="Открывает список занятий" onPress={props.onLesson} style={[styles.lesson, large && styles.largeLesson]} testID="home-lesson"><Text style={small}>Занятие</Text><Text style={[text, styles.bold]}>{props.lessonTitle ?? 'Выбери занятие'}</Text></Pressable>;
  const state = <View style={[styles.state, large && styles.largeState]} testID="home-care"><Text style={text}>{food}</Text><Text style={text}>{care}</Text><Text accessibilityLabel={`Настроение: ${mood}`} style={text}>{mood}</Text></View>;

  return <SafeAreaView style={styles.safe}>
    <View style={styles.home} onLayout={({ nativeEvent: { layout } }) => setSize(layout)} testID={large ? 'home-large-summary' : 'home-v42'}>
      <View pointerEvents="none" style={StyleSheet.absoluteFill} importantForAccessibility="no-hide-descendants">
        <FinniHomeScene key={`${profile.shapeId}-${profile.patternId}`} accessibilityLabel={`${profile.name}. ${food}. ${care}. ${mood}`} careLabel={`${food}. ${care}`} height={size.height} paused={props.scenePaused || menu || Boolean(props.notice)} motionEnabled={props.motionEnabled} stage={life.petStage} appearance={profile} reaction={props.reaction} onReactionFinished={props.onReactionFinished} onReactionCancelled={props.onReactionCancelled} showCaption={false} fullscreen petRegion={region} hideSkip skipReactionId={skipReactionId} />
      </View>
      <View testID="home-hud" style={[styles.hud, large && styles.largeHud, { height: size.height }]} accessibilityElementsHidden={menu} importantForAccessibility={menu ? 'no-hide-descendants' : 'auto'} pointerEvents={menu ? 'none' : 'box-none'}>
        {!large && <View style={styles.header}><Pressable accessibilityRole="button" accessibilityLabel={`О питомце: ${profile.name}`} onPress={props.onEditPet} style={styles.identity}><Text style={[styles.text, styles.bold]}>{profile.name}</Text><Text style={styles.small}>{model.dayLabel}{props.demo ? ' · Демо' : ''}</Text></Pressable><RouteButton label={canSkip ? "Пропустить" : "Как играть"} onPress={canSkip ? () => setSkipReactionId(props.reaction!.id) : props.onHelp} /></View>}
        <View style={[styles.summary, large && styles.largeSummary]} testID="home-finances">
          {large && <Text style={small}>Монеты</Text>}
          <View style={[styles.money, large && styles.largeMoney]}>{[['Доступно', life.available], ['Копилка', life.savings]].map(([label, value]) => <View key={label} accessible accessibilityLabel={`${label}: ${value} монет`} style={[styles.metric, large && styles.largeMetric]}><Text style={large ? text : small}>{large ? label : `${label}, монет`}</Text><Text style={[styles.amount, large && styles.largeText]}>{value}</Text></View>)}</View>
          <Pressable accessibilityRole="button" accessibilityHint="Открывает цель и накопления" onPress={() => props.onSection('Копилка')} style={[styles.goal, large && styles.largeGoal]} testID="home-goal">
            {large && <Text style={small}>Цель</Text>}<Text style={[text, styles.bold]}>{large ? goalName : `Цель: ${goalName}`}</Text>
            {goal && (large ? <><Text style={text}>{life.savings} / {goal.cost}</Text><Text style={text}>Осталось {remaining}</Text></> : <><Text style={small}>{life.savings} / {goal.cost} монет · Осталось {remaining}</Text><View accessibilityRole="progressbar" accessibilityValue={{ min: 0, max: goal.cost, now: Math.min(life.savings, goal.cost) }} style={styles.track}><View style={[styles.fill, { width: `${Math.min(100, life.savings / goal.cost * 100)}%` }]} /></View></>)}
          </Pressable>
        </View>
        {large ? <>
          <View style={styles.largePetRow} onLayout={({ nativeEvent: { layout } }) => setRegion({ x: layout.x + 2, y: layout.y, width: layout.width * .27, height: layout.height })}>
            <Pressable accessibilityRole="button" accessibilityLabel={`${profile.name}, стадия ${life.petStage}. Изменить питомца`} onPress={props.onEditPet} style={styles.petTarget} testID="home-pet-target" />{state}
          </View>{lesson}
          <View testID="home-footer" style={styles.largeFooter}><Pressable accessibilityRole="button" accessibilityLabel={canSkip ? "Пропустить анимацию" : "Разделы"} onPress={() => canSkip ? setSkipReactionId(props.reaction!.id) : setMenu(true)} style={styles.sections} testID="home-sections"><Text style={small}>{canSkip ? "Пропуск" : "Разделы"}</Text></Pressable>{button}</View>
        </> : <>
          <View style={styles.inlineState}>{state}</View>{lesson}
          <View style={styles.sceneSpace} onLayout={({ nativeEvent: { layout } }) => setRegion({ x: layout.x + 70, y: layout.y + 4, width: Math.max(80, layout.width - 140), height: layout.height - 8 })}>
            <Pressable accessibilityRole="button" accessibilityLabel={`${profile.name}, стадия ${life.petStage}. ${mood}. Изменить питомца`} onPress={props.onEditPet} style={styles.ordinaryPetTarget} />
            <View style={styles.rightRail}><RouteButton label="Прогресс" onPress={() => props.onSection('Прогресс')} /><RouteButton label="Для взрослого" onPress={() => props.onSection('Для взрослого')} /></View>
              <View pointerEvents="box-none" style={styles.objects}>{[
                ['OBJ-PLANNER', 'Планер', 'План'], ['OBJ-GOAL-DISPLAY', 'Цель', 'Копилка'], ['OBJ-CARE', 'Забота', 'Покупки'], ['OBJ-CHEST', 'Копилка', 'Копилка'],
              ].map(([id, label, route]) => <Pressable key={id} accessibilityRole="button" accessibilityLabel={label} onPress={() => props.onSection(route!)} style={[styles.object, id === 'OBJ-PLANNER' ? styles.plannerAnchor : id === 'OBJ-CARE' ? styles.careAnchor : id === 'OBJ-GOAL-DISPLAY' ? styles.goalAnchor : styles.chestAnchor]} testID={`room-object-${id}`}>
                {id === 'OBJ-PLANNER' && <View pointerEvents="none" style={styles.table}><View style={styles.tableLeftLeg} /><View style={styles.tableRightLeg} /></View>}
                {!failedObjects.includes(id!) && <Image accessible={false} source={id === 'OBJ-GOAL-DISPLAY' ? goalSource(goal?.id ?? null) ?? OBJECT_SOURCES[id!] : OBJECT_SOURCES[id!]} resizeMode="contain" onError={() => setFailedObjects((old) => [...old, id!])} style={[styles.objectImage, id === 'OBJ-PLANNER' && styles.plannerImage]} />}<Text style={[styles.objectLabel, id === 'OBJ-PLANNER' && styles.plannerLabel]}>{label}</Text>
              </Pressable>)}</View>
          </View>
          <View testID="home-footer" style={styles.footer}>{button}<View style={styles.nav} accessibilityRole="tablist">{['Домик', 'План', 'Покупки', 'Копилка'].map((label) => <Pressable key={label} accessibilityRole="tab" accessibilityState={{ selected: label === 'Домик' }} onPress={() => label !== 'Домик' && props.onSection(label)} style={styles.navItem}><Text style={[styles.small, label === 'Домик' && styles.bold]}>{label}</Text></Pressable>)}</View></View>
        </>}
      </View>
      <Modal visible={menu} onRequestClose={() => setMenu(false)} animationType="none" accessibilityViewIsModal>
        <SafeAreaView style={styles.menu}><ScrollView contentContainerStyle={styles.menuContent}><Text style={[styles.text, styles.bold]}>{profile.name}</Text><Text style={styles.text}>{model.dayLabel}{props.demo ? ' · Демо' : ''}</Text><RouteButton label="Вернуться в домик" onPress={() => setMenu(false)} />{['План', 'Покупки', 'Копилка', 'Прогресс', 'Для взрослого'].map((label) => <RouteButton key={label} label={label} onPress={() => navigate(() => props.onSection(label))} />)}<RouteButton label="Занятия" onPress={() => navigate(props.onLesson)} /><RouteButton label="О питомце" onPress={() => navigate(props.onEditPet)} /><RouteButton label="Как играть" onPress={() => navigate(props.onHelp)} /></ScrollView></SafeAreaView>
      </Modal>
      <Modal visible={Boolean(props.notice)} transparent onRequestClose={props.onDismissNotice}><View style={styles.noticeOverlay}><View style={styles.summary}><Text accessibilityLiveRegion="polite" style={styles.text}>{props.notice}</Text><RouteButton label="Понятно" onPress={props.onDismissNotice} /></View></View></Modal>
    </View>
  </SafeAreaView>;
}

const styles = StyleSheet.create({
  safe: { flex: 1, backgroundColor: '#DDE8DE' },
  home: { flex: 1 }, hud: { padding: 8, gap: 1 }, largeHud: { padding: 6, gap: 4 },
  text: { color: '#263E43', fontSize: 16, lineHeight: 18 }, largeText: { fontSize: 16, lineHeight: 17.3 }, small: { color: '#425A5A', fontSize: 12, lineHeight: 14 }, largeSmall: { fontSize: 12, lineHeight: 13 }, bold: { fontWeight: '700' },
  header: { flexDirection: 'row', minHeight: 48, gap: 8 }, identity: { flex: 1, justifyContent: 'center', minHeight: 48, paddingHorizontal: 8, backgroundColor: '#FFFAF0EE', borderRadius: 14 },
  summary: { backgroundColor: '#FFFAF0F2', padding: 8, borderRadius: 18 }, largeSummary: { padding: 6 }, money: { flexDirection: 'row', gap: 8 }, largeMoney: { flexDirection: 'column', gap: 0 }, metric: { flex: 1 }, largeMetric: { flexDirection: 'row', justifyContent: 'space-between', flex: 0, flexWrap: 'wrap', gap: 4 }, amount: { color: '#263E43', fontSize: 18, lineHeight: 21, fontWeight: '800', fontVariant: ['tabular-nums'] },
  goal: { minHeight: 48, paddingTop: 2 }, largeGoal: { paddingTop: 2 }, track: { height: 4, borderRadius: 2, backgroundColor: '#D8E3DA', marginTop: 2 }, fill: { height: 4, backgroundColor: '#4E786D', borderRadius: 2 },
  state: { flexDirection: 'row', flexWrap: 'wrap', gap: 6 }, inlineState: { borderRadius: 10, backgroundColor: '#FFFAF0E8', padding: 4 }, largeState: { flexDirection: 'column', flexWrap: 'nowrap', gap: 0, marginLeft: '30%', padding: 4, borderRadius: 14, backgroundColor: '#FFFAF0EE', flex: 1 },
  lesson: { backgroundColor: '#FFFAF0F2', borderRadius: 14, paddingHorizontal: 8, paddingVertical: 4, minHeight: 48, justifyContent: 'center' }, largeLesson: { padding: 4 },
  sceneSpace: { flex: 1, minHeight: 208 }, rightRail: { position: 'absolute', right: 0, top: 56, bottom: 0, width: 62, gap: 0 }, route: { backgroundColor: '#FFFAF0F2', borderRadius: 14, minHeight: 48, minWidth: 48, padding: 3, justifyContent: 'center' }, routeText: { fontSize: 12, lineHeight: 14, color: '#263E43', textAlign: 'center' }, pressed: { opacity: .75 },
  objects: { position: 'absolute', left: 0, top: 0, bottom: 0, right: 0 }, plannerAnchor: { left: 0, top: 28, height: 70 }, careAnchor: { left: 0, bottom: 0 }, goalAnchor: { right: 0, top: 0 }, chestAnchor: { right: 0, bottom: 0 }, object: { position: 'absolute', width: 62, height: 56, alignItems: 'center', justifyContent: 'center' }, objectImage: { width: 38, height: 38 }, objectLabel: { backgroundColor: '#FFFAF0E8', borderRadius: 4, color: '#263E43', fontSize: 12, lineHeight: 14, paddingHorizontal: 2 },
  table: { position: 'absolute', left: 5, right: 5, top: 36, height: 5, borderRadius: 2, backgroundColor: '#9C7047' }, tableLeftLeg: { position: 'absolute', left: 5, top: 5, height: 15, width: 4, backgroundColor: '#805737' }, tableRightLeg: { position: 'absolute', right: 5, top: 5, height: 15, width: 4, backgroundColor: '#805737' }, plannerImage: { position: 'absolute', top: 0 }, plannerLabel: { position: 'absolute', bottom: 0 },
  ordinaryPetTarget: { position: 'absolute', left: 70, top: 0, bottom: 0, right: 70 }, largePetRow: { minHeight: 100, flexDirection: 'row' }, petTarget: { position: 'absolute', left: 0, width: '27%', top: 0, bottom: 0, minHeight: 48, minWidth: 48 },
  footer: { gap: 6, height: 110, flexShrink: 0 }, primary: { backgroundColor: '#4E786D', borderRadius: 16, minHeight: 48, padding: 6, alignItems: 'center', justifyContent: 'center' }, primaryText: { color: '#FFFFFF', fontSize: 16, lineHeight: 18, fontWeight: '800', textAlign: 'center' }, disabled: { opacity: .55 }, nav: { height: 56, flexDirection: 'row', borderRadius: 18, backgroundColor: '#FFFAF0F2', padding: 4 }, navItem: { flex: 1, minHeight: 48, alignItems: 'center', justifyContent: 'center' }, largeFooter: { position: 'absolute', left: 6, right: 6, bottom: 6, flexDirection: 'row', gap: 4, height: 48 }, sections: { width: '34%', minHeight: 48, minWidth: 48, backgroundColor: '#FFFAF0F2', borderRadius: 14, justifyContent: 'center', alignItems: 'center', padding: 4 }, largePrimary: { flex: 1 },
  menu: { flex: 1, backgroundColor: '#EAF1E8' }, menuContent: { padding: 16, gap: 8 }, noticeOverlay: { flex: 1, justifyContent: 'center', padding: 24, backgroundColor: '#00000066' },
});
